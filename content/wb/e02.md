Taskflow is the second project people build when they learn Express: a **todo API**. You've written routes, validation and error handlers in Greenhouse, so they're already in the starting code. This project is about what changes when an API has to **keep** its data. The data has to survive a restart and a power cut, it has to grow past one screen, and it has to show up safely on a page.

Your tasks live in a file, **tasks.json**, which you'll see in the **Disk** panel as it changes. Once there's a file, the terminal grows a **⚡ pull the plug** button: it kills the server in the middle of a write, the way a power cut would.

// client.js: the terminal. It starts your server, then turns each line you type into an HTTP request
// and sends it with fetch(), the way a browser, curl or another service would. You don't need to change it.
//
//   GET /notes                         a request
//   POST /notes {"text": "milk"}       with a JSON body
//   GET /notes -i                      -i shows the response's headers too
//   GET /me -H "Authorization: abc"    -H adds a request header
require("./app");

const PORT = 3000;
// headers Node adds to every response, left out so the ones your code sets stand out
const HIDDEN = new Set(["connection", "content-length", "date", "keep-alive", "transfer-encoding"]);

function parse(line) {
  const m = /^(\S+)\s+(\S+)\s*(.*)$/.exec(line.trim());
  if (!m) return null;
  let [, method, path, rest] = m;
  const headers = {};
  let verbose = false;
  rest = rest.replace(/(^|\s)-H\s+"([^"]*)"/g, (_, space, h) => {
    const i = h.indexOf(":");
    headers[h.slice(0, i).trim()] = h.slice(i + 1).trim();
    return space;
  });
  rest = rest.replace(/(^|\s)-i(?=\s|$)/g, () => { verbose = true; return ""; }).trim();
  if (rest && !Object.keys(headers).some((k) => k.toLowerCase() === "content-type")) {
    headers["Content-Type"] = /^[[{]/.test(rest) ? "application/json" : "text/plain";
  }
  return { method: method.toUpperCase(), path, headers, body: rest || undefined, verbose };
}

async function handle(line) {
  if (!line.trim()) return null;
  const r = parse(line);
  if (!r || !r.path.startsWith("/")) return 'usage: METHOD /path [body] [-i] [-H "Name: value"]';
  let res;
  try {
    res = await fetch(`http://localhost:${PORT}${r.path}`, {
      method: r.method, headers: r.headers, body: r.body, redirect: "manual", signal: AbortSignal.timeout(2000),
    });
  } catch (err) {
    if (err.name === "TimeoutError") return "(no response in 2 s: the server never finished its answer)";
    return `(no response: ${err.cause ? err.cause.message : err.message})`;
  }
  const out = [`${res.status} ${res.statusText}`];
  if (r.verbose) for (const [k, v] of res.headers) if (!HIDDEN.has(k)) out.push(`${k}: ${v}`);
  const text = await res.text();
  if (text) out.push(...(r.verbose ? ["", text] : [text]));
  return out.join("\n");
}

module.exports = { handle };

---brief
A todo list is for ticking things off. Add:

- **`PATCH /api/tasks/:id`**: change only the fields that were sent, `title` (trimmed), `priority` and `done`. Check them with `problem(body, true)` first. `{"done": true}` ticks a task off; `{"done": "yes"}` is a 400, because `done` must be a real `true` or `false`.
- **`DELETE /api/tasks/:id`**: remove it and answer **204**.

Both use `findTask`, so a missing task is a 404 for free.
---why
PATCH is the natural verb for *"mark this done"*: the client sends the one field that changed. Checking types strictly matters more than it looks. `"false"` as a string is truthy in JavaScript, so a filter on `if (task.done)` would treat it as done. Refusing anything but a real boolean at the door keeps bad data out of the file you're about to start saving.

---brief
Press **↻ restart** in the terminal: every task is gone. They only ever lived in memory. Time to keep them in a file.

Write **store.js**, a small module that owns the file **tasks.json**:

- **`load()`**: if the file exists, read it and `JSON.parse` it; if not, return `[]`.
- **`save(tasks)`**: write `JSON.stringify(tasks)` and a newline. Keep it on **one line**, with no spacing.

Then in app.js, start with `const tasks = store.load()`, and call **`store.save(tasks)`** after every change, *before* answering. Ids must carry on after a restart: `nextId` is one more than the highest id loaded, not 1.
---why
Putting all the file handling in one module, behind `load` and `save`, means app.js never touches `fs`. When you swap the file for a real database later, only store.js changes. Saving *before* answering is a promise: once a client sees `201`, the task is on disk. Answer first, crash before the save, and the client was told a lie. The id rule matters too: restart at 1 and you'd hand out an id that's already taken.

// store.js: keeps the tasks in a file, tasks.json, so they outlive the server.
// app.js never touches fs: everything about the file is in here.
const fs = require("fs");

const FILE = "tasks.json";

function load() {
  if (!fs.existsSync(FILE)) return [];
  return JSON.parse(fs.readFileSync(FILE, "utf8"));
}

function save(tasks) {
  fs.writeFileSync(FILE, JSON.stringify(tasks) + "\n");
}

module.exports = { load, save };

// app.js never touches fs: everything about the file is in here.
const fs = require("fs");

const FILE = "tasks.json";

function load() {
  if (!fs.existsSync(FILE)) return [];
  try {
    return JSON.parse(fs.readFileSync(FILE, "utf8"));
  } catch (err) {
    // a torn or hand-edited file: keep it for a human, and start empty rather than not at all
    fs.renameSync(FILE, FILE + ".corrupt");
    console.error(`${FILE} couldn't be read, so it was moved to ${FILE}.corrupt and Taskflow started empty`);
    return [];
  }
}

// a crash leaves the old file or the new one, never half of one
function save(tasks) {
  fs.writeFileSync(FILE + ".tmp", JSON.stringify(tasks) + "\n");
  fs.renameSync(FILE + ".tmp", FILE);
}

title: Sorting and stats
files: app.js
---brief
Two more things every list app needs:

- **`?sort=title`** sorts A to Z, and **`?sort=priority`** puts high first, then normal, then low, keeping ties in id order. Sorting happens **after** filtering and **before** paging, so page 2 is page 2 of the sorted list. Sort a copy, never `tasks` itself.
- **`GET /api/stats`** answers the totals: `{"total": 4, "done": 1, "open": 3, "byPriority": {"high": 2, "normal": 1, "low": 1}}`.
---why
The order of the steps is the whole lesson: **filter, then sort, then page**. Page before sorting and every page is sorted on its own, so a high-priority task on page 3 never reaches page 1. Databases follow the same order: `WHERE`, then `ORDER BY`, then `LIMIT`/`OFFSET`. Priorities sort by a rank (`{high: 0, normal: 1, low: 2}`), not alphabetically, which would put `high` before `low` before `normal`.
---picture
  GET /api/tasks?done=false&sort=priority&page=2&limit=5

  tasks ─▶ filter (done=false) ─▶ sort (high, normal, low) ─▶ page 2 of 5 ─▶ res.json
           like SQL:  WHERE           ORDER BY                  LIMIT 5 OFFSET 5

// app.js: Taskflow, a todo API. client.js starts it with require("./app").
const express = require("express");
const store = require("./store");

const app = express();
app.use(express.json());

const PRIORITIES = ["low", "normal", "high"];
// the tasks live in tasks.json; this array is the copy in memory, saved after every change
const tasks = store.load();
let nextId = tasks.reduce((max, t) => Math.max(max, t.id), 0) + 1;

// an error that carries the status to answer with, for the error handler at the bottom
function httpError(status, message) {
  const err = new Error(message);
  err.status = status;
  return err;
}

// what's wrong with a task a client sent, or null. partial (PATCH): only check the fields it sent
function problem(body, partial = false) {
  if (!partial || "title" in body) {
    if (typeof body.title !== "string" || !body.title.trim()) return "title is required";
  }
  if ("priority" in body && !PRIORITIES.includes(body.priority)) return "priority must be low, normal or high";
  if ("done" in body && typeof body.done !== "boolean") return "done must be true or false";
  return null;
}

// middleware: loads the task named by :id into req.task
function findTask(req, res, next) {
  req.task = tasks.find((t) => t.id === Number(req.params.id));
  if (!req.task) throw httpError(404, "Task not found");
  next();
}

// priorities sort by rank, not alphabetically
const RANK = { high: 0, normal: 1, low: 2 };

// GET /api/tasks?done=false&priority=high&q=bulbs&sort=priority&page=2&limit=5
app.get("/api/tasks", (req, res) => {
  let result = tasks;
  if (req.query.done !== undefined) {
    if (req.query.done !== "true" && req.query.done !== "false") throw httpError(400, "done must be true or false");
    result = result.filter((t) => t.done === (req.query.done === "true"));
  }
  if (req.query.priority) result = result.filter((t) => t.priority === req.query.priority);
  if (req.query.q) {
    const text = req.query.q.toLowerCase();
    result = result.filter((t) => t.title.toLowerCase().includes(text));
  }
  if (req.query.sort === "title") result = [...result].sort((a, b) => a.title.localeCompare(b.title));
  else if (req.query.sort === "priority") result = [...result].sort((a, b) => RANK[a.priority] - RANK[b.priority] || a.id - b.id);
  const page = Number(req.query.page || 1), limit = Number(req.query.limit || 10);
  if (!Number.isInteger(page) || !Number.isInteger(limit) || page < 1 || limit < 1) {
    throw httpError(400, "page and limit must be whole numbers from 1");
  }
  const size = Math.min(limit, 50), pages = Math.ceil(result.length / size);
  // the paging goes in headers, so the body stays the same plain array it always was
  res.set("X-Total-Count", String(result.length));
  const link = (p, rel) => {
    const q = new URLSearchParams(req.query);
    q.set("page", p);
    q.set("limit", size);
    return `<${req.path}?${q}>; rel="${rel}"`;
  };
  const links = [];
  if (page > 1) links.push(link(page - 1, "prev"));
  if (page < pages) links.push(link(page + 1, "next"));
  if (links.length) res.set("Link", links.join(", "));
  res.json(result.slice((page - 1) * size, page * size));
});

app.get("/api/stats", (req, res) => {
  const byPriority = { high: 0, normal: 0, low: 0 };
  for (const t of tasks) byPriority[t.priority]++;
  const done = tasks.filter((t) => t.done).length;
  res.json({ total: tasks.length, done, open: tasks.length - done, byPriority });
});

app.get("/api/tasks/:id", findTask, (req, res) => {
  res.json(req.task);
});

app.post("/api/tasks", (req, res) => {
  const body = req.body || {};
  const bad = problem(body);
  if (bad) throw httpError(400, bad);
  const task = { id: nextId++, title: body.title.trim(), priority: body.priority || "normal", done: false };
  tasks.push(task);
  store.save(tasks);
  res.status(201).location(`/api/tasks/${task.id}`).json(task);
});

app.patch("/api/tasks/:id", findTask, (req, res) => {
  const body = req.body || {};
  const bad = problem(body, true);
  if (bad) throw httpError(400, bad);
  if ("title" in body) req.task.title = body.title.trim();
  if ("priority" in body) req.task.priority = body.priority;
  if ("done" in body) req.task.done = body.done;
  store.save(tasks);
  res.json(req.task);
});

app.delete("/api/tasks/:id", findTask, (req, res) => {
  tasks.splice(tasks.indexOf(req.task), 1);
  store.save(tasks);
  res.status(204).end();
});

// every error ends up here, as JSON
app.use((err, req, res, next) => {
  if (err.type === "entity.parse.failed") return res.status(400).json({ error: "Invalid JSON" });
  const status = err.status || 500;
  if (status >= 500) console.error(`${req.method} ${req.originalUrl} failed: ${err.message}`);
  res.status(status).json({ error: status >= 500 ? "Something went wrong" : err.message });
});

app.listen(3000, () => console.log("Taskflow on http://localhost:3000"));

id: t8
title: A page of its own
files: app.js
---brief
APIs are read by programs, but people like a page. Answer **`GET /`** with HTML built from the tasks, the way Express apps did long before front-end frameworks:

~~~html
<!doctype html><title>Taskflow</title><h1>Taskflow</h1><p>1 of 2 done</p><ul><li class="done normal">Buy bulbs</li>…</ul>
~~~

Each task is an `<li>` whose class is `done` or `open` and then its priority. **Escape every title** before it goes into the HTML. Write `escapeHtml(text)` to turn `&`, `<`, `>`, `"` and `'` into `&amp;`, `&lt;`, `&gt;`, `&quot;` and `&#39;`. Then try adding a task titled `<script>alert('hi')</script>`.
---why
Without escaping, a task's title becomes part of your page's code. Anyone who can add a task can then run JavaScript in everyone's browser: stealing sessions, posting as them. That's **cross-site scripting** (XSS), one of the most common security bugs on the web. The rule is simple and absolute: text from users is escaped at the moment it goes into HTML. Template engines like EJS and Pug, and React, do this for you, which is exactly why you should understand what they're doing.
---picture
  title:    <script>alert('hi')</script>
  escaped:  &lt;script&gt;alert(&#39;hi&#39;)&lt;/script&gt;      shown as text, never run
---todo
- `GET /` answers an HTML page built from the tasks
- `escapeHtml` handles all five characters
- Every title is escaped
