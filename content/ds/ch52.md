```prelude
import time

def timed(fn, *args):
    t = time.perf_counter()
    out = fn(*args)
    return out, time.perf_counter() - t

def lcg(n, seed, mx):
    out, x = [], seed
    for _ in range(n):
        x = (x * 1103515245 + 12345) % 2 ** 31
        out.append(x % mx)
    return out

def road_map(n, extra, seed):
    """A connected map: a random tree on 0..n-1 plus about `extra` random two-way roads."""
    g = [x / 10 ** 9 for x in lcg(n + 2 * extra, seed, 10 ** 9)]
    edges = [(int(g[v] * v), v) for v in range(1, n)]
    for i in range(extra):
        a, b = int(g[n + 2 * i] * n), int(g[n + 2 * i + 1] * n)
        if a != b:
            edges.append((a, b))
    return edges
```

## Single points of failure

In a network of two-way links (roads, power lines, network cables), a **bridge** is a link whose loss splits the network, and an **articulation point** (cut vertex) is a node whose loss does. Finding them tells you where to add redundancy.

The obvious way: remove each edge, search, see if everything is still reachable. That's m searches, O(m·(n + m)), far too slow for 10⁵ roads.

**Tarjan's low-link** runs one DFS and gives every node two numbers:

- `disc[v]`: the time the search first reached v (0, 1, 2…).
- `low[v]`: the smallest `disc` that v's part of the tree can reach using tree edges downwards **plus one back edge** (an edge to a node already visited).

When the search returns from child c to parent p, set `low[p] = min(low[p], low[c])`, and test:

- **Bridge.** Edge p–c is a bridge when `low[c] > disc[p]`: nothing in c's subtree can get back to p or above without that edge.
- **Articulation point.** A non-root p is a cut vertex when some child has `low[c] ≥ disc[p]`: that subtree can reach p, but not above it, so removing p strands it. The **root** is a cut vertex only when it has two or more DFS children.

Two traps. Skip the edge you came in on by its **id**, not by "is it my parent?": two parallel roads between the same towns mean neither is a bridge, and skipping by node would miss the second one. And use an explicit stack, because 10⁵ nodes overflow Python's recursion.

```run
title: Tarjan's low-link, iterative
def low_link(n, edges):
    adj = [[] for _ in range(n)]
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
        adj[b].append((a, i))
    disc, low, t = [-1] * n, [0] * n, 0
    bridges, cuts = [], set()
    for s in range(n):
        if disc[s] != -1:
            continue
        disc[s] = low[s] = t
        t += 1
        kids, st = 0, [[s, -1, 0]]          # node, edge we came in on, next neighbour
        while st:
            top = st[-1]
            v, pe, i = top
            if i < len(adj[v]):
                top[2] += 1
                w, eid = adj[v][i]
                if eid == pe:
                    continue
                if disc[w] == -1:
                    disc[w] = low[w] = t
                    t += 1
                    st.append([w, eid, 0])
                    if v == s:
                        kids += 1
                elif disc[w] < low[v]:
                    low[v] = disc[w]
            else:
                st.pop()
                if st:
                    p = st[-1][0]
                    low[p] = min(low[p], low[v])
                    if low[v] > disc[p]:
                        bridges.append(pe)
                    if p != s and low[v] >= disc[p]:
                        cuts.add(p)
        if kids > 1:
            cuts.add(s)
    return bridges, cuts

E = [(0, 1), (1, 2), (2, 0), (2, 3), (3, 4), (4, 5), (5, 3), (5, 6), (6, 7), (7, 8), (8, 9), (9, 7), (4, 10), (10, 11), (1, 3)]
br, cuts = low_link(12, E)
print(sorted(E[i] for i in br), sorted(cuts))
```

```exercise

---solution
def roads_to_add(n, edges):
    adj = [[] for _ in range(n)]
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
        adj[b].append((a, i))
    disc, low, t, br = [-1] * n, [0] * n, 0, set()
    for s in range(n):
        if disc[s] != -1:
            continue
        disc[s] = low[s] = t
        t += 1
        st = [[s, -1, 0]]
        while st:
            top = st[-1]
            v, pe, i = top
            if i < len(adj[v]):
                top[2] += 1
                w, eid = adj[v][i]
                if eid == pe:
                    continue
                if disc[w] == -1:
                    disc[w] = low[w] = t
                    t += 1
                    st.append([w, eid, 0])
                elif disc[w] < low[v]:
                    low[v] = disc[w]
            else:
                st.pop()
                if st:
                    p = st[-1][0]
                    low[p] = min(low[p], low[v])
                    if low[v] > disc[p]:
                        br.add(pe)
    par = list(range(n))
    def find(x):
        while par[x] != x:
            par[x] = par[par[x]]
            x = par[x]
        return x
    for i, (a, b) in enumerate(edges):
        if i not in br:
            par[find(a)] = find(b)
    deg = {}
    for i in br:
        a, b = edges[i]
        for r in (find(a), find(b)):
            deg[r] = deg.get(r, 0) + 1
    leaves = sum(1 for d in deg.values() if d == 1)
    return (leaves + 1) // 2
```

```exercise
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
```

```exercise
def worst_blackout(n, edges):
    adj = [[] for _ in range(n)]
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
        adj[b].append((a, i))
    disc, low, sub = [-1] * n, [0] * n, [1] * n
    disc[0], t, best = 0, 1, 0
    st = [[0, -1, 0]]
    while st:
        top = st[-1]
        v, pe, i = top
        if i < len(adj[v]):
            top[2] += 1
            w, eid = adj[v][i]
            if eid == pe:
                continue
            if disc[w] == -1:
                disc[w] = low[w] = t
                t += 1
                st.append([w, eid, 0])
            elif disc[w] < low[v]:
                low[v] = disc[w]
        else:
            st.pop()

                p = st[-1][0]
                low[p] = min(low[p], low[v])
                sub[p] += sub[v]
                if low[v] > disc[p]:
```

## Rebuild it and explain it

```exercise
title: lowlink_rebuild
kind: rebuild
---prompt
From memory: `low_link(n, edges)` returning `(bridges, cuts)`: a sorted list of bridge edge ids, and a sorted list of articulation points. One iterative DFS.
---starter

---solution
def low_link(n, edges):
    adj = [[] for _ in range(n)]
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
        adj[b].append((a, i))
    disc, low, t = [-1] * n, [0] * n, 0
    bridges, cuts = [], set()
    for s in range(n):
        if disc[s] != -1:
            continue
        disc[s] = low[s] = t
        t += 1
        kids, st = 0, [[s, -1, 0]]
        while st:
            top = st[-1]
            v, pe, i = top
            if i < len(adj[v]):
                top[2] += 1
                w, eid = adj[v][i]
                if eid == pe:
                    continue
                if disc[w] == -1:
                    disc[w] = low[w] = t
                    t += 1
                    st.append([w, eid, 0])
                    if v == s:
                        kids += 1
                elif disc[w] < low[v]:
                    low[v] = disc[w]
            else:
                st.pop()
                if st:
                    p = st[-1][0]
                    low[p] = min(low[p], low[v])
                    if low[v] > disc[p]:
                        bridges.append(pe)
                    if p != s and low[v] >= disc[p]:
                        cuts.add(p)
        if kids > 1:
            cuts.add(s)
    return sorted(bridges), sorted(cuts)
```

```recall
Q: Explain Tarjan's low-link for bridges and articulation points: what disc and low mean, the two tests and the root special case, why you skip the parent edge by id, and how bridges lead to "fewest new roads to remove every bridge".
A: One DFS gives every node disc, the time it was first reached, and low, the smallest disc its subtree can reach using tree edges downwards plus one back edge. When the search returns from child c to parent p, set low[p] = min(low[p], low[c]); a back edge to an already-visited w sets low[v] = min(low[v], disc[w]). Tree edge p–c is a bridge when low[c] > disc[p], meaning nothing below c can climb back to p or above without it. A non-root p is an articulation point when some child has low[c] ≥ disc[p]. The root has nothing above it, so it's a cut vertex only when it has two or more DFS children. Skip only the edge you arrived by, compared by id: a second parallel edge to the parent is a real back edge, and two parallel edges are never bridges. Use an explicit stack for large graphs; the whole thing is O(n + m). Merging the towns joined by non-bridge edges gives 2-edge-connected blocks, which form a tree with the bridges. Each leaf block needs another way out and one new road can serve two leaves, so the fewest roads that remove every bridge is (leaves + 1) // 2.
```
